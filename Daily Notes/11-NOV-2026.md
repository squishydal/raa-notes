`bivent.service` this would be your bivent.service and you need to change user and group to root
```
> cat bivent.service
  [Unit]
Description=Bivent backend service
After=network.target docker.service
Requires=docker.service

[Service]
Type=simple
User=sharma << THIS ONE 
Group=sharma << AND THIS ONE
WorkingDirectory=/opt/bivent/etc
ExecStart=/usr/bin/bivent -c bivent.yaml web
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target
```
